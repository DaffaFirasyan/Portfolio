# Star Map Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let the reader explore the portfolio as a star map they operate (drag, zoom, click a constellation), with the current linear page kept as a "Reading mode" switch.

**Architecture:** The existing `SkyCanvas` gains a camera mode. A pure `mapEngine` holds the camera maths and the six constellations; a `ViewProvider` holds the mode, the open region (mirrored in the URL hash) and the camera refs; `StarMap` handles input; `MapPanel` wraps the existing section components as a drawer or sheet. Sections stay mounted and are hidden when inactive.

**Tech Stack:** Vite 8, React 19, TypeScript 6, Tailwind 4, Vitest + Testing Library + jsdom. No new dependency.

Spec: `docs/superpowers/specs/2026-11-20-star-map-design.md`.

```ts
import { hash, seeded } from '@/sky/engine';

export const WORLD = { w: 3000, h: 2000 } as const;
export const MIN_ZOOM = 0.2;
export const MAX_ZOOM = 3;
export const FLY_MS = 650;
export const PANEL_W = 720;

export const REGION_IDS = ['about', 'skills', 'experience', 'projects', 'education', 'contact'] as const;
export type RegionId = (typeof REGION_IDS)[number];

export interface Camera {
  /** The world point at the centre of the screen. */
  x: number;
  y: number;
  zoom: number;
}
export interface View {
  w: number;
  h: number;
}
export interface Region {
  id: RegionId;
  x: number;
  y: number;
  /** Star offsets from the region's centre, in world units. */
  stars: Array<[number, number]>;
}
export interface Motion {
  /** Screen pixels per millisecond, left over from a drag. */
  vx: number;
  vy: number;
  fly: { from: Camera; to: Camera; at: number; ms: number } | null;
}

const CENTER = { x: 1500, y: 1000 };
const RING = [470, 420, 480, 400, 470, 420];

/**
 * A small constellation: a random walk of `steps` segments, the same for the
 * same id, centred on its own middle. No two ids look alike.
 */
export function figure(id: string, steps = 5, near = 70, far = 120): Array<[number, number]> {
  const rand = seeded(hash(`${id}#figure`));
  const points: Array<[number, number]> = [[0, 0]];
  let angle = rand() * Math.PI * 2;
  for (let i = 0; i < steps; i++) {
    angle += (rand() - 0.5) * 2.2;
    const step = near + rand() * (far - near);
    const [px, py] = points[points.length - 1];
    points.push([px + Math.cos(angle) * step, py + Math.sin(angle) * step]);
  }
  const cx = points.reduce((sum, p) => sum + p[0], 0) / points.length;
  const cy = points.reduce((sum, p) => sum + p[1], 0) / points.length;
  return points.map(([x, y]) => [x - cx, y - cy]);
}

/** Six constellations on a ring round the middle of the world, starting at the top. */
export const REGIONS: Region[] = REGION_IDS.map((id, i) => {
  const angle = ((-90 + i * 60) * Math.PI) / 180;
  return {
    id,
    x: CENTER.x + Math.cos(angle) * RING[i],
    y: CENTER.y + Math.sin(angle) * RING[i],
    stars: figure(id),
  };
});

export const clampZoom = (zoom: number) => Math.min(MAX_ZOOM, Math.max(MIN_ZOOM, zoom));

export function toWorld(cam: Camera, view: View, sx: number, sy: number): [number, number] {
  return [(sx - view.w / 2) / cam.zoom + cam.x, (sy - view.h / 2) / cam.zoom + cam.y];
}

export function toScreen(cam: Camera, view: View, wx: number, wy: number): [number, number] {
  return [(wx - cam.x) * cam.zoom + view.w / 2, (wy - cam.y) * cam.zoom + view.h / 2];
}

export function clampCamera(cam: Camera): Camera {
  return {
    zoom: clampZoom(cam.zoom),
    x: Math.min(WORLD.w, Math.max(0, cam.x)),
    y: Math.min(WORLD.h, Math.max(0, cam.y)),
  };
}

/** Zoom by `factor` keeping the world point under the pointer where it is. */
export function zoomAt(cam: Camera, view: View, sx: number, sy: number, factor: number): Camera {
  const [wx, wy] = toWorld(cam, view, sx, sy);
  const zoom = clampZoom(cam.zoom * factor);
  return clampCamera({
    zoom,
    x: wx - (sx - view.w / 2) / zoom,
    y: wy - (sy - view.h / 2) / zoom,
  });
}

/** Drag the sky by (dx, dy) screen pixels. */
export function panBy(cam: Camera, dx: number, dy: number): Camera {
  return clampCamera({ ...cam, x: cam.x - dx / cam.zoom, y: cam.y - dy / cam.zoom });
}

export const easeInOut = (t: number) => (t < 0.5 ? 4 * t * t * t : 1 - (-2 * t + 2) ** 3 / 2);

export function flyStep(from: Camera, to: Camera, t: number): Camera {
  const e = easeInOut(Math.min(1, Math.max(0, t)));
  return {
    x: from.x + (to.x - from.x) * e,
    y: from.y + (to.y - from.y) * e,
    zoom: from.zoom * (to.zoom / from.zoom) ** e,
  };
}

const decay = (v: number, dt: number) => v * Math.exp(-dt / 320);

/**
 * One frame of camera motion: a flight if there is one, else the coast left over
 * from a drag. Mutates `motion`, returns the next camera (the same object when idle).
 */
export function stepCamera(cam: Camera, motion: Motion, now: number, dt: number): Camera {
  const fly = motion.fly;
  if (fly) {
    const t = (now - fly.at) / fly.ms;
    if (t >= 1) {
      motion.fly = null;
      return { ...fly.to };
    }
    return flyStep(fly.from, fly.to, t);
  }
  if (Math.abs(motion.vx) < 0.01 && Math.abs(motion.vy) < 0.01) {
    motion.vx = 0;
    motion.vy = 0;
    return cam;
  }
  const next = panBy(cam, motion.vx * dt, motion.vy * dt);
  motion.vx = decay(motion.vx, dt);
  motion.vy = decay(motion.vy, dt);
  return next;
}

/** The constellation nearest a world point, if it is within `radius` world units. */
export function hitRegion(regions: Region[], wx: number, wy: number, radius = 130): Region | null {
  let best: Region | null = null;
  let bestDistance = radius;
  for (const region of regions) {
    const distance = Math.hypot(region.x - wx, region.y - wy);
    if (distance <= bestDistance) {
      best = region;
      bestDistance = distance;
    }
  }
  return best;
}

export const panelWidth = (view: View) => (view.w >= 1024 ? PANEL_W : 0);

/** A camera that puts `region` in the middle of the part of the screen the panel leaves free. */
export function cameraFor(region: Region, view: View, zoom = 1.6): Camera {
  const free = view.w - panelWidth(view);
  return clampCamera({ zoom, x: region.x - (free / 2 - view.w / 2) / zoom, y: region.y });
}

/** Every constellation on screen, with the middle of the ring pushed away from the title block. */
export function overviewCamera(view: View): Camera {
  const zoom = clampZoom(Math.min(view.w / 1400, (view.h - 140) / 1400) * 0.95);
  const wide = view.w >= 1024;
  return clampCamera({
    zoom,
    x: CENTER.x - (wide ? (0.14 * view.w) / zoom : 0),
    y: CENTER.y - (wide ? 0 : (0.1 * view.h) / zoom),
  });
}

/** Where a region's label button goes: centred under its figure. */
export function labelPosition(region: Region, cam: Camera, view: View): [number, number] {
  const [sx, sy] = toScreen(cam, view, region.x, region.y);
  return [sx, sy + Math.min(150, 130 * cam.zoom) + 8];
}
```

```tsx
export type Mode = 'map' | 'read';
export const VIEW_KEY = 'daffa_view';

interface View {
  mode: Mode;
  /** False when there is no 2D canvas: the map cannot draw, so the page is always Reading. */
  canMap: boolean;
  setMode: (mode: Mode) => void;
  /** The constellation whose section is open in the panel, from the URL hash. */
  region: RegionId | null;
  open: (id: RegionId) => void;
  close: () => void;
  /** Move the camera: a flight where motion is allowed, a cut where it is not. */
  flyTo: (target: Camera) => void;
  camera: MutableRefObject<Camera>;
  motion: MutableRefObject<Motion>;
  /** Set by the sky: repaint a still frame. */
  paint: MutableRefObject<() => void>;
  /** Set by the map: put the region labels where the camera says. */
  layout: MutableRefObject<() => void>;
  /** Set by the sky: false in the still tier, where the camera cuts instead of flying. */
  animated: MutableRefObject<boolean>;
}

const INERT: View = {
  mode: 'read',
  canMap: false,
  setMode: () => {},
  region: null,
  open: () => {},
  close: () => {},
  flyTo: () => {},
  camera: { current: { x: 1500, y: 1000, zoom: 1 } },
  motion: { current: { vx: 0, vy: 0, fly: null } },
  paint: { current: () => {} },
  layout: { current: () => {} },
  animated: { current: true },
};

const Context = createContext<View>(INERT);

function canvasSupported(): boolean {
  try {
    return document.createElement('canvas').getContext('2d') !== null;
  } catch {
    return false;
  }
}

function loadMode(): Mode {
  try {
    return localStorage.getItem(VIEW_KEY) === 'read' ? 'read' : 'map';
  } catch {
    return 'map';
  }
}

export function regionFromHash(hash: string): RegionId | null {
  const id = hash.replace(/^#/, '');
  return (REGION_IDS as readonly string[]).includes(id) ? (id as RegionId) : null;
}

/**
 * Which way the page is read: as a map you move through, or as the plain page.
 * The open region lives in the URL hash, so a link, a reload and the Back button
 * all work. Like SkillHighlight and the passport it is inert outside a provider.
 */
export function ViewProvider({ children }: { children: ReactNode }) {
  const [canMap] = useState(canvasSupported);
  const [mode, setModeState] = useState<Mode>(() => (canMap ? loadMode() : 'read'));
  const [region, setRegion] = useState<RegionId | null>(() => regionFromHash(window.location.hash));
  const camera = useRef<Camera>(overviewCamera({ w: window.innerWidth, h: window.innerHeight }));
  const motion = useRef<Motion>({ vx: 0, vy: 0, fly: null });
  const paint = useRef(() => {});
  const layout = useRef(() => {});
  const animated = useRef(true);
  const scrollTo = useRef<string | null>(null);

  useEffect(() => {
    const sync = () => setRegion(regionFromHash(window.location.hash));
    window.addEventListener('hashchange', sync);
    window.addEventListener('popstate', sync);
    return () => {
      window.removeEventListener('hashchange', sync);
      window.removeEventListener('popstate', sync);
    };
  }, []);

  // The map is one screen: nothing behind it should scroll.
  useEffect(() => {
    if (mode !== 'map') return;
    const previous = document.body.style.overflow;
    document.body.style.overflow = 'hidden';
    return () => {
      document.body.style.overflow = previous;
    };
  }, [mode]);

  // Leaving the map with a panel open lands the reader on that section.
  useEffect(() => {
    if (mode === 'read' && scrollTo.current) {
      document.getElementById(scrollTo.current)?.scrollIntoView();
      scrollTo.current = null;
    }
  }, [mode]);

  const flyTo = useCallback((target: Camera) => {
    if (animated.current) {
      motion.current = {
        vx: 0,
        vy: 0,
        fly: { from: { ...camera.current }, to: target, at: performance.now(), ms: FLY_MS },
      };
    } else {
      motion.current = { vx: 0, vy: 0, fly: null };
      camera.current = target;
      paint.current();
    }
  }, []);

  const open = useCallback((id: RegionId) => {
    setRegion(id);
    if (window.location.hash !== `#${id}`) window.location.hash = id;
  }, []);

  const close = useCallback(() => {
    setRegion(null);
    if (window.location.hash) {
      window.history.pushState(null, '', window.location.pathname + window.location.search);
    }
  }, []);

  const setMode = useCallback(
    (next: Mode) => {
      if (next === 'map' && !canMap) return;
      try {
        localStorage.setItem(VIEW_KEY, next);
      } catch {
        // Storage is blocked: the choice lasts for this visit.
      }
      if (next === 'read') scrollTo.current = region;
      else setRegion(regionFromHash(window.location.hash));
      setModeState(next);
    },
    [canMap, region],
  );

  const value = useMemo<View>(
    () => ({ mode, canMap, setMode, region, open, close, flyTo, camera, motion, paint, layout, animated }),
    [mode, canMap, setMode, region, open, close, flyTo],
  );

  return <Context.Provider value={value}>{children}</Context.Provider>;
}

export function useView(): View {
  return useContext(Context);
}
```

```tsx
import { useEffect, useRef, type ReactNode } from 'react';

import { useLanguage } from '@/context/LanguageContext';
import { useView } from './ViewProvider';

/**
 * Wraps the sections. In Reading mode it is `display: contents`, so the page is
 * exactly the page it was; in Map mode it is a drawer on the right (a full-height
 * sheet on a phone) that slides in when a region is open. It is always the same
 * element, so switching mode never remounts a section and loses its filter.
 *
 * Opening it moves focus to the back button; closing it (the button, Escape or
 * the browser's Back) returns focus to the region's label, which `StarMap` does.
 * An open certificate dialog keeps Escape for itself.
 */
export default function MapPanel({ children }: { children: ReactNode }) {
  const { mode, region, close } = useView();
  const { t } = useLanguage();
  const back = useRef<HTMLButtonElement>(null);
  const map = mode === 'map';
  const open = map && region !== null;

  useEffect(() => {
    if (open) back.current?.focus();
  }, [open]);

  useEffect(() => {
```

```tsx
import { useEffect, useRef } from 'react';

import { usePassport } from '@/passport/PassportProvider';
import { DWELL_MS, type StampId } from '@/passport/stamps';
import { useView } from './ViewProvider';

/**
 * The reading rule for Map mode, which has no scrolling to measure: a region is
 * stamped once its panel has been open for five seconds in total and the reader
 * then leaves it (closes it, opens another, or switches to Reading mode). The
 * overview counts as "home". A hop through the map earns nothing.
 */
export default function MapStamps() {
  const { mode, region } = useView();
  const { earn } = usePassport();
  const spent = useRef<Record<string, number>>({});

  useEffect(() => {
    if (mode !== 'map') return;
    const id: StampId = region ?? 'home';
    const book = spent.current;
    const start = Date.now();
    return () => {
      book[id] = (book[id] ?? 0) + Date.now() - start;
      if (book[id] >= DWELL_MS) earn(id);
    };
  }, [mode, region, earn]);

  return null;
}
```

```tsx
import { useEffect, useRef } from 'react';

import { useLanguage } from '@/context/LanguageContext';
import {
  REGIONS,
  cameraFor,
  hitRegion,
  labelPosition,
  overviewCamera,
  panBy,
  toScreen,
  toWorld,
  zoomAt,
  type Region,
  type RegionId,
} from './mapEngine';
import { useView } from './ViewProvider';

const DRAG_PX = 6;
const KEY_STEP = 90;

const viewport = () => ({ w: window.innerWidth, h: window.innerHeight });

/**
 * The map's hands. A full-screen surface turns a drag into panning (with a
 * coast after you let go), the wheel and a two-finger pinch into zooming about
 * the pointer, a tap on a constellation into opening it, and the arrow keys and
 * plus and minus into the same for the keyboard. Each constellation also has a
 * real button, named by its section, so the map works without a pointer.
 *
 * The camera follows the open region: open one and it flies there, leaving the
 * free side of the screen to the panel; close it and it flies back to the
 * overview. A drag that moved never counts as a click.
 */
export default function StarMap() {
  const { sections } = useLanguage();
  const { mode, region, open, flyTo, camera, motion, paint, layout, animated } = useView();
  const surface = useRef<HTMLDivElement>(null);
  const labels = useRef(new Map<RegionId, HTMLButtonElement>());
  const moved = useRef(false);
  const previous = useRef<RegionId | null>(null);

  // The camera follows the region.
  useEffect(() => {
    if (mode !== 'map') return;
    const view = viewport();
    const target = REGIONS.find((r) => r.id === region);
    flyTo(target ? cameraFor(target, view, view.w >= 1024 ? 1.6 : 1.4) : overviewCamera(view));
    if (!region && previous.current) labels.current.get(previous.current)?.focus();
    previous.current = region;
  }, [mode, region, flyTo]);

  // The sky asks for the labels to be placed every frame.
  useEffect(() => {
    if (mode !== 'map') return;
    const place = () => {
      const view = viewport();
      for (const r of REGIONS) {
        const el = labels.current.get(r.id);
        if (!el) continue;
        const [x, y] = labelPosition(r, camera.current, view);
        el.style.transform = `translate(${x}px, ${y}px) translateX(-50%)`;
      }
    };
    layout.current = place;
    place();
    return () => {
      layout.current = () => {};
    };
  }, [mode, camera, layout]);

  // A resized window re-frames the current view without a flight.
  useEffect(() => {
    if (mode !== 'map') return;
    const onResize = () => {
      const view = viewport();
      const target = REGIONS.find((r) => r.id === region);
      motion.current = { vx: 0, vy: 0, fly: null };
      camera.current = target ? cameraFor(target, view, view.w >= 1024 ? 1.6 : 1.4) : overviewCamera(view);
      paint.current();
      layout.current();
    };
    window.addEventListener('resize', onResize);
    return () => window.removeEventListener('resize', onResize);
  }, [mode, region, camera, motion, paint, layout]);

  useEffect(() => {
    const el = surface.current;
    if (mode !== 'map' || !el) return;

    const pointers = new Map<number, { x: number; y: number }>();
    let pinch = 0;
    let startX = 0;
    let startY = 0;
    let lastAt = 0;
    const commit = () => {
      paint.current();
      layout.current();
    };
    const stop = () => {
      motion.current = { vx: 0, vy: 0, fly: null };
    };

    const onDown = (event: PointerEvent) => {
      pointers.set(event.pointerId, { x: event.clientX, y: event.clientY });
      if (pointers.size === 1) {
        moved.current = false;
        startX = event.clientX;
        startY = event.clientY;
        lastAt = event.timeStamp;
        stop();
      } else if (pointers.size === 2) {
        const [a, b] = [...pointers.values()];
        pinch = Math.hypot(a.x - b.x, a.y - b.y);
        moved.current = true;
      }
    };

    const onMove = (event: PointerEvent) => {
      const p = pointers.get(event.pointerId);
      if (!p) return;
      const dx = event.clientX - p.x;
      const dy = event.clientY - p.y;
      p.x = event.clientX;
      p.y = event.clientY;

      if (pointers.size === 2) {
        const [a, b] = [...pointers.values()];
        const dist = Math.hypot(a.x - b.x, a.y - b.y);
        if (pinch > 0) {
          camera.current = zoomAt(camera.current, viewport(), (a.x + b.x) / 2, (a.y + b.y) / 2, dist / pinch);
        }
        pinch = dist;
        commit();
        return;
      }
      if (!moved.current && Math.hypot(event.clientX - startX, event.clientY - startY) < DRAG_PX) return;
      moved.current = true;
      camera.current = panBy(camera.current, dx, dy);
      const dt = Math.max(1, event.timeStamp - lastAt);
      lastAt = event.timeStamp;
      motion.current.vx = dx / dt;
      motion.current.vy = dy / dt;
      commit();
    };

    const onUp = (event: PointerEvent) => {
      if (!pointers.delete(event.pointerId)) return;
      if (pointers.size < 2) pinch = 0;
      if (pointers.size > 0) return;
      if (!moved.current) {
        // A tap: open the constellation under it, with a finger-sized reach on a small screen.
        const [wx, wy] = toWorld(camera.current, viewport(), event.clientX, event.clientY);
        const hit = hitRegion(REGIONS, wx, wy, Math.max(130, 44 / camera.current.zoom));
        if (hit) open(hit.id);
        return;
      }
      // Coast only if the finger was still moving when it lifted, and only where motion is welcome.
      if (!animated.current || event.timeStamp - lastAt > 80) {
        motion.current.vx = 0;
        motion.current.vy = 0;
      }
    };

    const onWheel = (event: WheelEvent) => {
      event.preventDefault();
      stop();
      camera.current = zoomAt(
        camera.current,
        viewport(),
        event.clientX,
        event.clientY,
        Math.exp(-event.deltaY * 0.0015),
      );
      commit();
    };

    const onKey = (event: KeyboardEvent) => {
      const steps: Record<string, [number, number]> = {
        ArrowLeft: [KEY_STEP, 0],
        ArrowRight: [-KEY_STEP, 0],
        ArrowUp: [0, KEY_STEP],
        ArrowDown: [0, -KEY_STEP],
      };
      const view = viewport();
      if (event.key in steps) {
        stop();
        camera.current = panBy(camera.current, ...steps[event.key]);
      } else if (event.key === '+' || event.key === '=') {
        camera.current = zoomAt(camera.current, view, view.w / 2, view.h / 2, 1.2);
      } else if (event.key === '-') {
        camera.current = zoomAt(camera.current, view, view.w / 2, view.h / 2, 1 / 1.2);
      } else if (event.key === '0') {
        flyTo(overviewCamera(view));
        return;
      } else {
        return;
      }
      event.preventDefault();
      commit();
    };

    el.addEventListener('pointerdown', onDown);
    el.addEventListener('wheel', onWheel, { passive: false });
    el.addEventListener('keydown', onKey);
    window.addEventListener('pointermove', onMove);
    window.addEventListener('pointerup', onUp);
    window.addEventListener('pointercancel', onUp);
    return () => {
      el.removeEventListener('pointerdown', onDown);
      el.removeEventListener('wheel', onWheel);
      el.removeEventListener('keydown', onKey);
      window.removeEventListener('pointermove', onMove);
      window.removeEventListener('pointerup', onUp);
      window.removeEventListener('pointercancel', onUp);
    };
  }, [mode, open, flyTo, camera, motion, paint, layout, animated]);

  if (mode !== 'map') return null;

  const reveal = (r: Region) => {
    const [x, y] = toScreen(camera.current, viewport(), r.x, r.y);
    const { w, h } = viewport();
    if (x < 0 || x > w || y < 56 || y > h) flyTo({ ...camera.current, x: r.x, y: r.y });
  };

  return (
    <div
      ref={surface}
      data-map-surface
      tabIndex={-1}
      className="fixed inset-0 z-0 touch-none outline-none"
    >
      <div
        data-region-labels
        inert={region !== null}
        onClickCapture={(event) => {
          // The release of a drag is not a click on whatever label it ended over.
          if (moved.current) {
            event.preventDefault();
            event.stopPropagation();
          }
        }}
      >
        {REGIONS.map((r) => (
          <button
            key={r.id}
            ref={(el) => {
              if (el) labels.current.set(r.id, el);
              else labels.current.delete(r.id);
            }}
```

```ts
export interface ProjectFigure {
  id: string;
  /** World positions. */
  stars: Array<[number, number]>;
}

/** One small figure per project, set on a ring inside the Projects region. */
export function projectFigures(ids: string[], around: Region): ProjectFigure[] {
  return ids.map((id, i) => {
    const angle = ((45 + (360 / Math.max(ids.length, 1)) * i) * Math.PI) / 180;
    const cx = around.x + Math.cos(angle) * 170;
    const cy = around.y + Math.sin(angle) * 170;
    return {
      id,
      stars: figure(id, 3, 45, 70).map(([x, y]): [number, number] => [cx + x, cy + y]),
    };
  });
}

/**
 * Advance a trace along a figure's stars. `index` is the last star reached
 * (-1: not started). The pointer must come within `tol` of the next star; a
 * trace begins only at the first star and never skips one.
 */
export function traceStep(
  points: Array<[number, number]>,
  index: number,
  px: number,
  py: number,
  tol = 28,
): number {
  const next = index + 1;
  if (next >= points.length) return index;
  const [x, y] = points[next];
  return Math.hypot(x - px, y - py) <= tol ? next : index;
}
```
